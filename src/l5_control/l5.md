pitch_rate、yaw_acceleration、pitch_acceleration 和 shoot。前六项在串口
`TxPayload` 中均按 `float` 编码，单位依次为 rad、rad、rad/s、rad/s、
rad/s²、rad/s²。命令超时后保留最后角度，速度、加速度和 shoot 清零。

struct AimReference  // 非MPC模式的瞄准参考
  {
      int target_id{-1};           // 当前锁定的目标车辆ID，-1表示未锁定

struct FireConfig //判断标准
{
  // 第一阶段必须保持 false；完成验收后由实车配置显式修改。
  bool shoot_enable{false};                 //开枪状态

  std::optional<double> bullet_diameter;  //meter; 17 mm projectile = 0.017  
  std::optional<double> min_bullet_speed;  // meter per second   最小弹速
  std::optional<double> max_bullet_speed;  // meter per second   
  std::optional<double> heat_limit;// 热量限制

  std::optional<double> min_yaw;   //目标装甲板最小 yaw（机械范围）
  std::optional<double> max_yaw;    //目标装甲板最大 yaw
  std::optional<double> min_pitch;   //目标装甲板最小 pitch
  std::optional<double> max_pitch;   //目标装甲板最大 pitch
  std::optional<double> max_aim_yaw_error;   //目标装甲板最大 yaw_error(判断云台是否对准目标)
  std::optional<double> max_aim_pitch_error;   //目标装甲板最大 pitch_error
  std::optional<double> confidence_threshold;   //目标装甲板最小置信度(判断云台是否对准目标)
  std::optional<double> max_yaw_command_jump;       // yaw命令最大允许跳变量
  std::optional<double> max_pitch_command_jump;     // pitch命令最大允许跳变量
  std::optional<double> max_yaw_error;              // 云台就位时允许的yaw误差
  std::optional<double> max_pitch_error;            // 云台就位时允许的pitch误差    
  std::chrono::milliseconds max_robot_state_age{50};  // 机器人状态最长留存时间
  std::chrono::milliseconds max_gimbal_pose_age{20};// 云台姿态最长留存时间

  [[nodiscard]] bool parametersReady() const noexcept   //判断所需参数是否准备好了
  {
    return bullet_diameter.has_value() && min_bullet_speed.has_value() &&
           max_bullet_speed.has_value() && heat_limit.has_value() &&
           min_yaw.has_value() && max_yaw.has_value() &&
           min_pitch.has_value() && max_pitch.has_value() &&
           max_aim_yaw_error.has_value() && max_aim_pitch_error.has_value() &&
           confidence_threshold.has_value() &&
           max_yaw_command_jump.has_value() && max_pitch_command_jump.has_value() &&
           max_yaw_error.has_value() && max_pitch_error.has_value();
  }
};

3.若可信的话 目标所处状态是否符合开火条件
判断yaw pitch 和v 等等是否符合fireconfig所在范围（包含机械范围和误差范围）
由状态机判断是否处于稳定追踪状态

4.若目标状态稳定 云台能否稳定的移动到开火的位置
云台是否就位（判断yaw pitch误差是否在允许范围内）
 yaw或者pitch是否发生突变（对比两者差值，超过阈值则认为发生突变）
（发送给云台的角度命令是否出现跳变情况）
是否处于切板的状态(通过和上一次的数据所对比，相同则没切板，不同则切板)
